        function setVehicle(type, price) {
            currentVehicle = type;
            baseFare = price;
            document.querySelectorAll('.veh-card').forEach(c => c.classList.remove('active-veh'));
            document.getElementById('veh-' + type).classList.add('active-veh');
            recalc();
        }

        function setZone(zone) {
            currentZone = zone;
            document.getElementById('tab-city').classList.toggle('active', zone === 'city');
            document.getElementById('tab-out').classList.toggle('active', zone === 'out');
            document.getElementById('out-controls').style.opacity = zone === 'city' ? '0.35' : '1';
            document.getElementById('out-controls').style.pointerEvents = zone === 'city' ? 'none' : 'auto';
            recalc();
        }

        function setKm(val) {
            currentKm = Math.max(1, Math.min(250, parseInt(val) || 1));
            document.getElementById('km-display').innerText = currentKm;
            document.getElementById('km-range').value = Math.min(150, currentKm);

            document.querySelectorAll('.preset-chip').forEach(chip => {
                chip.classList.toggle('active', chip.innerText.includes(`(${currentKm} km)`));
            });

            if (currentZone !== 'out') setZone('out');
            else recalc();
        }

        function recalc() {
            const total = currentZone === 'city' ? baseFare : baseFare + (currentKm * PER_KM);
            document.getElementById('total-mdl').innerText = total.toLocaleString('ru-RU');
        }

        function detectGPS() {
            const btnText = document.getElementById('gps-btn-text');
            const locInput = document.getElementById('sos-loc');

            if (!navigator.geolocation) {
                alert(lang === 'ru' ? 'Геолокация не поддерживается вашим браузером' : 'Geolocația nu este suportată de browser');
                return;
            }

            btnText.innerText = lang === 'ru' ? 'Определяем...' : 'Se caută...';

            navigator.geolocation.getCurrentPosition(
                (pos) => {
                    const lat = pos.coords.latitude.toFixed(5);
                    const lng = pos.coords.longitude.toFixed(5);
                    locInput.value = `GPS: https://maps.google.com/?q=${lat},${lng}`;
                    btnText.innerText = lang === 'ru' ? 'Координаты получены ✓' : 'Locație adăugată ✓';
                },
                () => {
                    btnText.innerText = dict[lang].gpsBtn;
                    alert(lang === 'ru' ? 'Разрешите доступ к GPS или напишите ориентир вручную.' : 'Permiteți accesul GPS sau scrieți adresa manual.');
                }
            );
        }

        function setLang(l) {
            lang = l;
            localStorage.setItem('ecoburn_lang', l);
            document.documentElement.lang = l;
            const t = dict[l];

            document.querySelectorAll('[data-i18n]').forEach(el => {
                const key = el.getAttribute('data-i18n');
                if (t[key]) el.innerHTML = t[key];
            });

            document.getElementById('sos-loc').placeholder = t.phLoc;
            document.getElementById('sos-car').placeholder = t.phCar;

            document.getElementById('btn-ru').classList.toggle('active', l === 'ru');
            document.getElementById('btn-ro').classList.toggle('active', l === 'ro');
            recalc();
        }

        function sendSOS(platform) {
            const loc = document.getElementById('sos-loc').value.trim();
            const car = document.getElementById('sos-car').value.trim();
            const total = document.getElementById('total-mdl').innerText;

            if (!loc || !car) {
                alert(lang === 'ru' ? 'Укажите, где вы находитесь и марку авто.' : 'Indicați locația dvs. și marca automobilului.');
                return;
            }

            const routeInfo = currentZone === 'city'
                ? (lang === 'ru' ? 'По г. Бельцы' : 'În mun. Bălți')
                : `${currentKm} km (${total} MDL)`;

            const msg = lang === 'ru'
                ? `🚨 СРОЧНО НУЖЕН ЭВАКУАТОР!\n📍 Где: ${loc}\n🚗 Авто/Проблема: ${car}\n🛣️ Маршрут: ${routeInfo}`
                : `🚨 AM NEVOIE URGENTĂ DE EVACUATOR!\n📍 Locația: ${loc}\n🚗 Mașina/Problema: ${car}\n🛣️ Traseu: ${routeInfo}`;

            const phone = "37378200474";
            const url = platform === 'whatsapp'
                ? `https://wa.me/${phone}?text=${encodeURIComponent(msg)}`
                : `viber://chat?number=%2B${phone}&draft=${encodeURIComponent(msg)}`;

            window.open(url, '_blank');
        }

        // Subtle Amber Highway Light Streaks Canvas
        const canvas = document.getElementById('road-canvas');
        const ctx = canvas.getContext('2d');
        let streaks = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        const particleCount = window.innerWidth < 640 ? 14 : 26;
        for (let i = 0; i < particleCount; i++) {
            streaks.push({
                x: Math.random() * canvas.width,
                y: Math.random() * canvas.height,
                len: Math.random() * 24 + 8,
                vy: Math.random() * 1.1 + 0.35,
                alpha: Math.random() * 0.38 + 0.1
            });
        }

        function drawStreaks() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            streaks.forEach(s => {
                s.y -= s.vy;
                if (s.y < -30) {
                    s.y = canvas.height + 30;
                    s.x = Math.random() * canvas.width;
                }
                ctx.beginPath();
                ctx.moveTo(s.x, s.y);
                ctx.lineTo(s.x, s.y + s.len);
                ctx.strokeStyle = `rgba(255, 183, 0, ${s.alpha})`;
                ctx.lineWidth = 1.6;
                ctx.stroke();
            });
            requestAnimationFrame(drawStreaks);
        }
        drawStreaks();

        // Init
        setLang(lang);
